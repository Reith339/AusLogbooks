cmake_minimum_required(VERSION 3.21)
project(LogBook VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)

find_package(Qt6 REQUIRED COMPONENTS Widgets Sql Network)

# Google OAuth client for the Google Sheets backup (a "Desktop app" client from Google Cloud).
# Provide it with -DGOOGLE_CLIENT_ID=... -DGOOGLE_CLIENT_SECRET=..., or in a git-ignored
# google-oauth.cmake file next to this one (see google-oauth.cmake.example).
# Without it the app builds fine; the Google backup option just shows as unavailable.
set(GOOGLE_CLIENT_ID "" CACHE STRING "Google OAuth client ID (Desktop app)")
set(GOOGLE_CLIENT_SECRET "" CACHE STRING "Google OAuth client secret (Desktop app)")
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/google-oauth.cmake)
    include(${CMAKE_CURRENT_SOURCE_DIR}/google-oauth.cmake)
endif()

set(SOURCES
    src/main.cpp
    src/Columns.h          src/Columns.cpp
    src/TimeUtil.h         src/TimeUtil.cpp
    src/Database.h         src/Database.cpp
    src/Recency.h          src/Recency.cpp
    src/FlightTableModel.h src/FlightTableModel.cpp
    src/FlightDialog.h     src/FlightDialog.cpp
    src/AircraftDialog.h   src/AircraftDialog.cpp
    src/BroughtForwardDialog.h src/BroughtForwardDialog.cpp
    src/TotalsWidget.h     src/TotalsWidget.cpp
    src/RecencyWidget.h    src/RecencyWidget.cpp
    src/PageWidget.h       src/PageWidget.cpp
    src/Theme.h            src/Theme.cpp
    src/SettingsDialog.h   src/SettingsDialog.cpp
    src/SheetFormat.h      src/SheetFormat.cpp
    src/GoogleAuth.h       src/GoogleAuth.cpp
    src/GoogleSheetsBackup.h src/GoogleSheetsBackup.cpp
    src/GoogleBackupDialog.h src/GoogleBackupDialog.cpp
    src/MainWindow.h       src/MainWindow.cpp
)

if(WIN32)
    # Exe icon + version info shown in Explorer.
    configure_file(resources/LogBook.rc.in ${CMAKE_CURRENT_BINARY_DIR}/LogBook.rc @ONLY)
    list(APPEND SOURCES ${CMAKE_CURRENT_BINARY_DIR}/LogBook.rc)
endif()

if(APPLE)
    # LogBook.icns is generated by the macOS build (see packaging/build-macos.sh).
    set(ICNS ${CMAKE_CURRENT_SOURCE_DIR}/resources/LogBook.icns)
    if(EXISTS ${ICNS})
        list(APPEND SOURCES ${ICNS})
        set_source_files_properties(${ICNS} PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    endif()
endif()

add_executable(LogBook WIN32 MACOSX_BUNDLE ${SOURCES})

# Window icon (all platforms).
qt_add_resources(LogBook "icons" PREFIX "/" BASE resources FILES resources/icon_256.png)

target_compile_definitions(LogBook PRIVATE
    APP_VERSION="${PROJECT_VERSION}"
    GOOGLE_CLIENT_ID="${GOOGLE_CLIENT_ID}"
    GOOGLE_CLIENT_SECRET="${GOOGLE_CLIENT_SECRET}")
if(GOOGLE_CLIENT_ID)
    message(STATUS "Google Sheets backup: enabled")
else()
    message(STATUS "Google Sheets backup: no GOOGLE_CLIENT_ID set, feature disabled in this build")
endif()
# Read by the packaging scripts.
file(WRITE ${CMAKE_BINARY_DIR}/version.txt "${PROJECT_VERSION}")
target_link_libraries(LogBook PRIVATE Qt6::Widgets Qt6::Sql Qt6::Network)

set_target_properties(LogBook PROPERTIES
    OUTPUT_NAME "LogBook"
    MACOSX_BUNDLE_BUNDLE_NAME "Pilot Logbook"
    MACOSX_BUNDLE_GUI_IDENTIFIER "app.pilotlogbook.LogBook"
    MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
    MACOSX_BUNDLE_ICON_FILE "LogBook.icns"
    MACOSX_BUNDLE_COPYRIGHT "Pilot Logbook"
)
